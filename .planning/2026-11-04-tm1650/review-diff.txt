+static uint16_t flash_record_crc(const uint16_t *record)
+{
+	uint16_t crc=0xffffu;
+	uint8_t word;
+	uint8_t byte;
+	uint8_t bit;
+	/* 中文：CRC16-CCITT，按每半字低字节在先覆盖版本至32位序号，标志不参与。 */
+	for(word=1;word<7;word++)

+			}
+		}
+	}
